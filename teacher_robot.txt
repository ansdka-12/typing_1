  <script type="module">
    import { initializeApp } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-app.js";
    import { getFirestore, collection, onSnapshot, query, orderBy } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-firestore.js";

    const firebaseConfig = {
      apiKey: "AIzaSyCbD3IZgib2KCETLMqP2zMH2_WVVdxfsmk",
      authDomain: "test-26-01-20.firebaseapp.com",
      projectId: "test-26-01-20",
      storageBucket: "test-26-01-20.firebasestorage.app",
      messagingSenderId: "1046847215998",
      appId: "1:1046847215998:web:f9ce56e45c28c4e61f7b2f"
    };

    const app = initializeApp(firebaseConfig);
    const db = getFirestore(app);

    let allDocsData = [];
    const dateFilterSelect = document.getElementById("dateFilter");
    const q = query(collection(db, "typing_records"), orderBy("cpm", "desc"));

    onSnapshot(q, (snapshot) => {
      allDocsData = [];
      const datesSet = new Set();

      snapshot.forEach((doc) => {
        const data = doc.data();
        // ★ [핵심] '로봇' 교실 데이터만 필터링해서 가져옴
        if (data.roomName === "로봇") {
          let dateStr = "날짜 없음";
          if (data.timestamp) {
            const dateObj = data.timestamp.toDate ? data.timestamp.toDate() : new Date(data.timestamp);
            dateStr = `${dateObj.getFullYear()}-${String(dateObj.getMonth() + 1).padStart(2, '0')}-${String(dateObj.getDate()).padStart(2, '0')}`;
            datesSet.add(dateStr);
          }
          allDocsData.push({ ...data, dateStr });
        }
      });

      const selectedDate = dateFilterSelect.value;
      dateFilterSelect.innerHTML = '<option value="ALL">전체 누적 보기</option>';
      Array.from(datesSet).sort().reverse().forEach(date => {
        const option = document.createElement("option");
        option.value = date; option.innerText = date;
        if (date === selectedDate) option.selected = true;
        dateFilterSelect.appendChild(option);
      });

      renderTable();
    });

    dateFilterSelect.addEventListener("change", renderTable);

    function renderTable() {
      const selectedDate = dateFilterSelect.value;
      const tbody = document.getElementById("recordTable");
      tbody.innerHTML = "";
      let active = 0, complete = 0, rank = 1;

      allDocsData.forEach((data) => {
        if (selectedDate !== "ALL" && data.dateStr !== selectedDate) return;
        if (data.status === "진행 중") active++;
        if (data.status === "완료") complete++;

        let formattedTime = '-';
        if (data.elapsedTime !== undefined && data.elapsedTime !== null) {
          const min = Math.floor(data.elapsedTime / 60);
          const sec = data.elapsedTime % 60;
          formattedTime = min > 0 ? `${min}분 ${sec}초` : `${sec}초`;
        }

        const tr = document.createElement("tr");
        tr.innerHTML = `
          <td>${data.status === '완료' ? rank++ : '-'}</td>
          <td>${data.dateStr}</td>
          <td>${data.studentId || '-'}</td>
          <td>${data.name}</td>
          <td>${formattedTime}</td>
          <td>${data.accuracy !== undefined ? data.accuracy + '%' : '-'}</td>
          <td>${data.cpm ? data.cpm + ' CPM' : '-'}</td>
        `;
        tbody.appendChild(tr);
      });

      document.getElementById("activeCount").innerText = active;
      document.getElementById("completeCount").innerText = complete;
    }
